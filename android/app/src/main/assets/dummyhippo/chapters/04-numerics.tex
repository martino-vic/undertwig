\section{Discrete operators}
Define the second-difference matrix $T_{n}\in\R^{n\times n}$ by
\begin{equation}
(T_{n}v)_{i}
=
\frac{v_{i-1}-2v_{i}+v_{i+1}}{h^{2}},
\qquad
h=\frac{1}{n+1},
\end{equation}
with homogeneous Dirichlet values $v_{0}=v_{n+1}=0$.

\begin{align}
\lambda_{k}(T_{n})
&=
-\frac{4}{h^{2}}\sin^{2}\!\left(\frac{k\pi}{2(n+1)}\right),
&
k&=1,\dots,n,
\\
\norm{T_{n}^{-1}}_{2}
&=
\frac{1}{\abs{\lambda_{1}(T_{n})}}
=
\frac{h^{2}}{4\sin^{2}\!\bigl(\pi/(2(n+1))\bigr)}.
\end{align}

\section{Algorithm box}
\begin{hipponote}[title={Pseudo-code payload}]
\begin{enumerate}[leftmargin=*,label=\arabic*.]
  \item Assemble sparse residual $r = b - Ax$.
  \item Apply $m$ steps of GMRES with restart length $\ell$.
  \item Update $x\leftarrow x+\delta$ and recompute $r$.
  \item Stop when $\norm{r}_{2}\le\tau\norm{b}_{2}$ or the iteration budget
        $K_{\max}$ is exhausted.
\end{enumerate}
The same loop is written again symbolically as
\[
x_{k+1}
=
x_{k}
+
\arg\min_{z\in\mathcal{K}_{m}(A,r_{k})}
\norm{b-A(x_{k}+z)}_{2}.
\]
\end{hipponote}
