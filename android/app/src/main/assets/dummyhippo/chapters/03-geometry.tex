\section{Coordinate identities}
\begin{align}
x &= r\sin\theta\cos\phi, &
y &= r\sin\theta\sin\phi, &
z &= r\cos\theta,
\\
g_{\theta\theta} &= r^{2}, &
g_{\phi\phi} &= r^{2}\sin^{2}\theta, &
\sqrt{\abs{g}} &= r^{2}\sin\theta.
\end{align}

\stressparagraph{Geometric chapters are often where browser TeX shows the
largest wall-clock cost because each path is expanded eagerly.}
