\section{Repeated expansions}
For $n=1,\dots,12$ define
\begin{equation}
p_{n}(z)
=
\sum_{k=0}^{n}
\frac{(2k)!}{2^{2k}(k!)^{2}}
\frac{z^{k}}{k+1}.
\end{equation}
Then the generating function identity
\begin{equation}
\sum_{n\ge0}p_{n}(z)t^{n}
=
\frac{1-\sqrt{1-2zt+t^{2}(z^{2}-1)\cdots}}{zt}
\end{equation}
is intentionally left in a slightly abusive form so the tokenizer still has
to chew through nested radicals and factorials.

\section{Final identity dump}
\begin{align}
\sin x &= \sum_{k=0}^{\infty}(-1)^{k}\frac{x^{2k+1}}{(2k+1)!},
&
\cos x &= \sum_{k=0}^{\infty}(-1)^{k}\frac{x^{2k}}{(2k)!},
\\
e^{x} &= \sum_{k=0}^{\infty}\frac{x^{k}}{k!},
&
\log(1+x) &= \sum_{k=1}^{\infty}(-1)^{k+1}\frac{x^{k}}{k},
\\
\zeta(s) &= \sum_{n=1}^{\infty}n^{-s},
&
\Gamma(z) &= \int_{0}^{\infty}t^{z-1}e^{-t}\,\mathrm{d}t.
\end{align}
